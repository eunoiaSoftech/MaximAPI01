Create Procedure [dbo].[USP_ADMIN_ManageMenus_SaveUpdateView] 
(    
@MenuName nvarchar(100),    
@ControllerName nvarchar(100),    
@ActionName nvarchar(100),    
@PartyId nvarchar(100)=null,    
@MenuId int,    
@Status int    
)    
As    
Begin    
	Begin Try 
		declare @Id int=0   
		if(@MenuId=0)
			select @Id=Isnull(iPK_MenuId,0) from MENUMST where sCntrllrNm=@ControllerName and sActnMthd=@ActionName and sMenuName=@MenuName    
		if(@Id<>0) 
			Begin 
				select 1 as StatusCode , 'All ready Exist In table' as [Message]     
			End    
		else    
			Begin 
				if(@MenuId<>0) 
					Begin 
						Update MENUMST set sCntrllrNm=@ControllerName,sActnMthd=@ActionName,sMenuName=@MenuName,iStatus=@Status    
						where iPK_MenuId=@MenuId    
						select 1 as StatusCode , 'Update Succefully' as [Message]     
					End    
				else    
					Begin 
						insert into MENUMST(sCntrllrNm,sActnMthd,sMenuName,iStatus,dtCrtdDt)
						values(@ControllerName,@ActionName,@MenuName,@Status,GETDATE())    
						select 1 as StatusCode , 'Record Saved Successfully' as [Message]     
					End    
			End    
	End Try     
	begin catch       
		--EXEC [dbo].[USP_ADMIN_StoreProcedureErrorHandling_Save] 0,0,1 --To log Stored procedure errors    
		select -1 as StatusCode, 'Check Error - ' + cast(ERROR_Number() as varchar) + ' ' + ERROR_MESSAGE () as [Message],'' AS Body    
	end catch     
End

create Procedure [dbo].[USP_ADMIN_CreateNewDepartment_Save]

@DepartmentId int ,
@DepartmentName nvarchar(500) ,
@PartyId nvarchar(100)=null,
@CreatedBy nvarchar(500)=null,
@type nvarchar(100)
as
Begin
	if(@type = 'Insert')
		begin
			if exists(select * from DPETMST where sDeptName=@DepartmentName)
				begin
					select -1 as StatusCode , 'Details Already Exists!' as [Message]   
				end
				else
					begin
						insert into DPETMST (sDeptName,iFK_PartyId,iStts,dtCrtdDt,sCrtdBy)
						values(	@DepartmentName,@PartyId,1,getdate(),@CreatedBy)
						select 1 as StatusCode , 'Details Inserted Successfully!' as [Message]   
				end
		end

	if(@type = 'Update')
		begin
			Update DPETMST set sDeptName =	@DepartmentName where iPK_DeptId= @DepartmentId
			select 1 as StatusCode , 'Details Updated Successfully!' as [Message]   
		end
End

create Procedure [dbo].[USP_ADMIN_CreateNewGroup_SaveUpdate]      
@GroupId int ,    
@GroupName nvarchar(500) ,      
@MenuID int ,      
@SubmenuId int ,        
@PartyId nvarchar(100) ,      
@CreatedBy nvarchar(500),    
@type nvarchar(100)    
      
as      
Begin      
	if(@type = 'Insert')    
		begin 
		  if(@GroupId=0)    
			begin 
				set @GroupId = (select isnull(max(iPK_GrpId),0)+1 from GRPMST)    
			end    
		  if exists(select * from GRPMST where sGrpName=@GroupName and iMenuId=@MenuID and iSubMenuId=@SubmenuId and sPartyCode=@PartyId)      
			begin 
				select -1 as StatusCode , 'Details Already Exists!' as [Message]         
			end      
		  else      
			begin 
				insert into GRPMST (iPK_GrpId,sGrpName,iMenuId,iSubMenuId,iStatus,sPartyCode,dtCrtdDt,sCrtdBy)      
				values( @GroupId,@GroupName,@MenuID,@SubmenuId,1,@PartyId,getdate(),@CreatedBy)      
				select 1 as StatusCode , 'Record Saved Successfully!' as [Message]         
			end      
		end    
	if(@type = 'Update')    
		begin    
			Update GRPMST set sGrpName =@GroupName,iMenuId=@MenuID,iSubMenuId=@SubmenuId where biPK_MSTId= @GroupId    
			select 1 as StatusCode , 'Record Update Successfully!' as [Message]       
		 end    
End

create Procedure [dbo].[USP_ADMIN_GetDepartmentGroupRolesMapping_Select] --'W000001'
(  
@Id nvarchar(50)=null  
)  
AS  
Begin   
	select distinct M.*,D.sDeptName DepartmentName,G.sGrpName GroupName,R.sRolName RoleName,G.sPartyCode PartyId from ROLDEPT M   
	inner join DPETMST D on  M.iDeptId = D.iPK_DeptId   
	inner join GRPMST G on  M.iGrpId = G.iPK_GrpId  and (@Id is null or  G.sPartyCode=@Id)  
	inner join ROLEMST R on  M.iRoleId = R.iPk_RolId   
	  where M.iStatus!=2
End

	begin
		select 0 as Id, 'Select Menu' as Text
		union
		select  iPK_MenuId as Id, sMenuName as Text  from MENUMST where iStatus<>2
	end

if(@type='Submenus')
	begin
		select iDsplyOrdId as Id, sSubMnuName as Text,iSubMnuId as ID1 from SUBMENUMST where iFK_MnuId =cast(@menuId as int) and  iStatus<>2 order by iDsplyOrdId

		select iPK_DeptId as DepartmentID,sDeptName as DepartmentName,format(dtCrtdDt,'dd-MM-yyyy hh:mm:ss tt') as CreatedDate, iStts from DPETMST where iStts<>2

		from GRPMST g 
		left join MENUMST m on m.iPK_MenuId = g.iMenuId
		left join SUBMENUMST sm on sm.iSubMnuId = g.iSubMenuId
		where g.iStatus<>2 and (g.sPartyCode= @PartyId or @PartyId Is Null)
	end

		from GRPMST g 
		left join MENUMST m on m.iPK_MenuId = g.iMenuId
		left join SUBMENUMST sm on sm.iSubMnuId = g.iSubMenuId
		where biPK_MSTId = cast(@menuId as int) and g.iStatus<>2
	end

		union
		Select iPK_DeptId Id,sDeptName text from DPETMST where iStts=1

		Select 0 Id,'Select Group' text

		Select   iPK_GrpId Id,sGrpName text from GRPMST where iStatus=1 and (@PartyId is null or sPartyCode=@PartyId)

	 select ISnull(U.lPK_UsrPgRght,0) UserID
	 ,M.biPK_MSTId ID
	 ,M.iPK_GrpId GroupID
	 ,Menu.sMenuName MenuName
	 ,SubMenu.sSubMnuName SubMenuName
	 ,isnull(U.iAllowInst,0) perInsert
	 ,isnull(U.iAllowDelt,0) perDelete
	 ,isnull(U.iAllowEdit,0) perEdit
	 ,isnull(U.iAllowView,0) perView
	 ,isnull(U.iStatus,0) perStatus 
	 from [dbo].[GRPMST] M 
	 left join ROLDEPT Mapping on Mapping.iPK_RoleDeptId=@MappingId and Mapping.iGrpId= @GroupId
	 left join [dbo].[USRRGHTS] U on U.iFK_MapngId=@MappingId and U.iMstGrupId=M.biPK_MSTId
	 left join MENUMST Menu on Menu.iPK_MenuId=M.iMenuId
	 left join SUBMENUMST SubMenu on SubMenu.iSubMnuId=M.iSubMenuId
	 where M.iPK_GrpId=@GroupId
End

create Procedure [dbo].[USP_ADMIN_UserwiseRoleAccessRights_Save]
	 (
		@Type nvarchar(50),
		@MappingId int,
		@PermissionId int=0,
		@MstGroupID int,
		@status int,
		@PartyId nvarchar(50)=null
	 )

AS
Begin
if(@PermissionId=0)
	Begin
		select @PermissionId=IsNull(lPK_UsrPgRght,0) from [dbo].[USRRGHTS] where iFK_MapngId=@MappingId and iMstGrupId=@MstGroupID
		End
	Print(@PermissionId)

	if(@PermissionId=0)
		Begin
			insert into [dbo].[USRRGHTS] (iFK_MapngId,iMstGrupId,iAllowInst,iAllowDelt,iAllowEdit,iAllowView,iStatus,sFK_PrtyCode)
			values (@MappingId, @MstGroupID,0,0,0,0,0,@PartyId)
		End

	select @PermissionId=IsNull(lPK_UsrPgRght,0) from [dbo].[USRRGHTS] where iFK_MapngId=@MappingId and iMstGrupId=@MstGroupID
	if(@Type='InsertOperation')
		Begin
			Update [dbo].[USRRGHTS] set iAllowInst=@status where lPK_UsrPgRght=@PermissionId
		End
	if(@Type='UpdateOpertaion')
		Begin
			Update [dbo].[USRRGHTS] set iAllowEdit=@status where lPK_UsrPgRght=@PermissionId
		End
	if(@Type='DeleteOperation')
		Begin
			Update [dbo].[USRRGHTS] set iAllowDelt=@status where lPK_UsrPgRght=@PermissionId
		End
	if(@Type='ViewOperation')
		Begin
			Update [dbo].[USRRGHTS] set iAllowView=@status where lPK_UsrPgRght=@PermissionId
		End
	if(@Type='Status')
		Begin
			Update [dbo].[USRRGHTS] set iStatus=@status where lPK_UsrPgRght=@PermissionId
		End
	select 1 as StatusCode , 'Successfully Update!' as [Message] 
End
